### Use the operation reader

The journal is directly accepted by Commons `operation_activity`:

~~~javascript

  require('/absolute/commons/host/connected_operation_activity.cjs');
const native = JSON.parse(fs.readFileSync('/absolute/run/receiving/native-process.json', 'utf8'));
const result = analyzeOperationActivity({
  operations: [{id: native.operation_id, title: 'SCF current application and review'}],
  captures: [],
  native_records: [{
    id: native.record,
    observed_at: native.observed_at,
    source: {path: native.record},
    record: native
  }]
});
process.stdout.write(JSON.stringify(result, null, 2) + '\n');
~~~
