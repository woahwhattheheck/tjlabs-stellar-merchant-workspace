Use an immutable public repository commit when making the source lock:

~~~sh

  --repository /absolute/merchant-checkout --revision ACTUAL_FULL_COMMIT \
  --output /private/APPLICATION_SOURCE.json

python3 products/govhub/application/workspace/build_application.py build \
  --source-root /absolute/merchant-checkout/products/govhub/application \
  --manifest /private/APPLICATION_SOURCE.json \
  --output /private/GovHub-Complete-Application.zip

python3 products/govhub/application/workspace/build_application.py verify \
  /private/GovHub-Complete-Application.zip --expected-sha256 ACTUAL_WHOLE_ZIP_SHA256
~~~
